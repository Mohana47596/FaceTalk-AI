+        headMesh.material.color.setHex(0xffffff);
+        headMesh.material.needsUpdate = true;
+        console.log('[Avatar3D] Restored original head texture map.');
+      }
+    }
+  }, [headMesh, uploadedFaceInfo]);
+
   return (
     <group ref={group}>
       <primitive
         object={scene}
         position={[0, -1.0, 0]}
-        scale={0.01}
+        scale={1.0}
         rotation={[0, 0, 0]}
       />
+
+      {/* 3D Facial Feature Landmark Helpers - Hidden when face texture is applied */}
+      {!uploadedFaceInfo && (
+        <>
+          <mesh ref={noseHelperRef}>
+            <sphereGeometry args={[0.005, 8, 8]} />
+            <meshBasicMaterial color="#ff3333" depthTest={false} transparent opacity={0.9} />
+          </mesh>
+          <mesh ref={leftEyeHelperRef}>
+            <sphereGeometry args={[0.005, 8, 8]} />
+            <meshBasicMaterial color="#3333ff" depthTest={false} transparent opacity={0.9} />
+          </mesh>
+          <mesh ref={rightEyeHelperRef}>
+            <sphereGeometry args={[0.005, 8, 8]} />

+          <mesh ref={mouthHelperRef}>
+            <sphereGeometry args={[0.005, 8, 8]} />
+            <meshBasicMaterial color="#33ff33" depthTest={false} transparent opacity={0.9} />
+          </mesh>
+        </>
+      )}
+
+      {/* Debug Plane Geometry with faceTexture - Hidden when face texture is applied */}
+      {!uploadedFaceInfo && debugPlaneRef && (
+        <mesh ref={debugPlaneRef} scale={[fScale * 0.18, fScale * 0.18, 1.0]}>
+          <planeGeometry args={[1.0, 1.0]} />
+          {faceTexture ? (
+            <meshBasicMaterial map={faceTexture} transparent={true} opacity={1.0} side={THREE.DoubleSide} depthWrite={false} polygonOffset polygonOffsetFactor={-10} />
+          ) : (
+            <meshBasicMaterial color="#ff00ff" transparent={true} opacity={1.0} side={THREE.DoubleSide} depthWrite={false} polygonOffset polygonOffsetFactor={-10} />
+          )}
+        </mesh>

     </group>
   );
